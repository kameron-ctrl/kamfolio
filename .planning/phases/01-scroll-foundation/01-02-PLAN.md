    ```typescript
    /**
     * Phase 1 scroll store (D-07, D-08, D-09, D-10).
     *
     * Single source of truth for scroll state. LenisProvider writes; downstream
     * components read. R3F components MUST use `useScrollStore.getState()` inside
     * `useFrame` — never selector subscriptions — to avoid 60fps reconciler churn.
     *
     * `displayVelocity` is the lerped value downstream visual effects consume.
     * The lerp happens inside `set` (not in a separate RAF loop) so it advances
     * once per Lenis tick — co-located with state updates.
     */
    import { create } from "zustand";

    export type ScrollPhase = "vertical" | "transition" | "horizontal";

    export interface ScrollState {
      scrollY: number;
      velocity: number;
      displayVelocity: number;
      phase: ScrollPhase;
      progress: number;
      set: (
        partial: Partial<Omit<ScrollState, "set" | "displayVelocity">>,
      ) => void;
    }

    /**
     * Lerp factor for displayVelocity (D-08, A3 in RESEARCH.md).
     * Discretion range from CONTEXT.md is 0.1-0.2; 0.15 is the midpoint.
     * Tunable during execution if HUD shows lag during rapid direction changes.
     */
    const LERP_FACTOR = 0.15;

    export const useScrollStore = create<ScrollState>((set, get) => ({
      scrollY: 0,
      velocity: 0,
      displayVelocity: 0,
      phase: "vertical",
      progress: 0,
      set: (partial) => {
        const state = get();
        const targetVelocity = partial.velocity ?? state.velocity;
        const nextDisplayVelocity =
          state.displayVelocity + (targetVelocity - state.displayVelocity) * LERP_FACTOR;
        set({
          ...partial,
          displayVelocity: nextDisplayVelocity,
        });
      },
    }));

    /**
     * Phase + intra-zone progress computation (D-09).
     * Pure function — LenisProvider calls this inside the lenis.on('scroll') callback
     * and passes the result to useScrollStore.getState().set(...).
     *
     * Boundaries:
     *   vertical:   [0.00, 0.40)
     *   transition: [0.40, 0.55)
     *   horizontal: [0.55, 1.00]
     *
     * `progress` is normalized 0..1 within the current zone.
     */
    export function computePhase(rawProgress: number): {
      phase: ScrollPhase;
      progress: number;
    } {
      // Clamp out-of-range inputs (D-09 implicit — page can't scroll outside [0,1])
      if (rawProgress <= 0) return { phase: "vertical", progress: 0 };
      if (rawProgress >= 1) return { phase: "horizontal", progress: 1 };

      if (rawProgress < 0.4) {
        return { phase: "vertical", progress: rawProgress / 0.4 };
      }
      if (rawProgress < 0.55) {
        return { phase: "transition", progress: (rawProgress - 0.4) / 0.15 };
      }
      return { phase: "horizontal", progress: (rawProgress - 0.55) / 0.45 };
    }
    ```
